1. Open and inspect a table

Run TableExample01_OpenAndInspect. It loads Employees and prints metadata in the Immediate window.

Automatic range discovery

Option Explicit

Public Sub TableExample01b_AutoRange()
    On Error GoTo ErrorHandler
    Dim employees As SqlTable
    Set employees = New SqlTable
    Set employees = employees.AutoRange("Employees")
    MsgBox "Automatically loaded " & employees.RowCount & " data rows and " & employees.ColumnCount & " columns.", vbInformation

    Dim r As Object
    Set r = employees.FindFirst("Name", "=", "Ivan")
    If r Is Nothing Then
        Debug.Print "Ivan was not found."
    Else
        Debug.Print "Name: "; r("Name")
        Debug.Print "Department: "; r("Department")
        Debug.Print "Salary: "; r("Salary")
    End If

    Exit Sub
ErrorHandler:
    MsgBox "AutoRange failed: " & Err.Description, vbCritical
End Sub

This uses the current workbook, starts at A1, detects the last populated row and column, and treats row 1 as headers.

Option Explicit

Public Sub TableExample01_OpenAndInspect()
    On Error GoTo ErrorHandler
    Dim employees As SqlTable
    Set employees = New SqlTable
    Set employees = employees.OpenRange( _
        ThisWorkbook.Worksheets("Employees").Range("A1:E13"))

    Debug.Print "Headers: "; employees.HeaderCount
    Debug.Print "Data rows: "; employees.RowCount
    Debug.Print "Columns: "; employees.ColumnCount
    Debug.Print "Has Salary: "; employees.HasColumn("Salary")
    Debug.Print "Has column 4: "; employees.HasColumn(4)
    MsgBox "Metadata was printed in the Immediate window.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "TableExample01 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

-- Metadata and row-count operations are client-side table access.
SELECT COUNT(*) FROM Employees;
Expected metadata: 5 headers/columns and 12 data rows.

2. Read cells, rows and columns

Run TableExample02_ReadValues. Public row indexes start at 1 for the first data row; the header row is not counted.

Option Explicit

Public Sub TableExample02_ReadValues()
    On Error GoTo ErrorHandler
    Dim employees As SqlTable
    Dim rowArray As Variant
    Dim rowObject As Object
    Dim salaries As Variant
    Dim allData As Variant

    Set employees = New SqlTable
    Set employees = employees.OpenRange( _
        ThisWorkbook.Worksheets("Employees").Range("A1:E13"))

    Debug.Print "First name: "; employees.GetCell(1, "Name")
    Debug.Print "Fifth salary: "; employees.GetCell(5, "Salary")

    rowArray = employees.GetRowArray(1)
    Debug.Print "Row array name: "; rowArray(2)

    Set rowObject = employees.GetRowDictionary(1)
    Debug.Print "Row dictionary name: "; rowObject("Name")
    Debug.Print "Row dictionary salary: "; rowObject("Salary")

    salaries = employees.GetColumn("Salary")
    Debug.Print "First salary from column: "; salaries(1)

    allData = employees.GetAll()
    Debug.Print "Header from GetAll: "; allData(1, 2)
    MsgBox "Values were printed in the Immediate window.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "TableExample02 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Name FROM Employees;
SELECT Salary FROM Employees;
SELECT * FROM Employees;
The first data row is Ivan. The second column of a row array is Name; the fourth is Salary.

3. Find rows and test existence

Run TableExample03_FindRows. FindFirst and collection items are row dictionaries.

Option Explicit

Public Sub TableExample03_FindRows()
    On Error GoTo ErrorHandler
    Dim employees As SqlTable
    Dim firstEmployee As Object
    Dim matches As Collection
    Dim row As Object

    Set employees = New SqlTable
    Set employees = employees.OpenRange( _
        ThisWorkbook.Worksheets("Employees").Range("A1:E13"))

    Set firstEmployee = employees.FindFirst("EmployeeId", "=", 5)
    If Not firstEmployee Is Nothing Then
        Debug.Print "Employee 5: "; firstEmployee("Name")
    End If

    Set matches = employees.FindAll("Department", "=", "IT")
    Debug.Print "IT matches: "; matches.Count
    For Each row In matches
        Debug.Print row("Name"); " - "; row("Salary")
    Next row

    Debug.Print "Viktor exists: "; employees.Exists("Name", "=", "Viktor")
    Debug.Print "Nobody exists: "; employees.Exists("Name", "=", "Nobody")
    MsgBox "Find results were printed in the Immediate window.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "TableExample03 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT * FROM Employees WHERE EmployeeId = 5 LIMIT 1;
SELECT * FROM Employees WHERE Department = 'IT';
SELECT EXISTS(SELECT 1 FROM Employees WHERE Name = 'Viktor');
Employee 5 is Elena. There are six IT matches. Viktor exists; Nobody does not.

4. Convert SqlTable to SqlQuery

Run TableExample04_QueryAndWrite. This is the recommended way to move from simple table access to joins, typed filters, ordering, and output.

Option Explicit

Public Sub TableExample04_QueryAndWrite()
    On Error GoTo ErrorHandler
    Dim employees As SqlTable
    Dim query As SqlQuery

    Set employees = New SqlTable
    Set employees = employees.OpenRange( _
        ThisWorkbook.Worksheets("Employees").Range("A1:E13"))

    Set query = employees.Query()
    Set query = query.SelectColumns("Name", "Salary")
    Set query = query.Where("Department", "=", "IT")
    Set query = query.OrderBy("Salary", "DESC")
    Set query = query.Limit(3)

    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "SqlTable query result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "TableExample04 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Name, Salary
FROM Employees
WHERE Department = 'IT'
ORDER BY Salary DESC
LIMIT 3;
Expected rows: Elena/4200, Ivan/3500, Ivan/3500.

Recommended relationship

Use SqlTable for direct reads and simple lookups. Use SqlTable.Query() when the operation becomes a query:

SqlTable → Query() → SqlQuery → ToArray()/WriteToRange()
