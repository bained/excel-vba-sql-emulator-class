General Examples

The complete, explanatory examples are listed in the Contents above.

1. Basic range query

Run Example01_BasicRangeQuery. It filters IT employees with salary above 2,000 and writes the result to Results.

Option Explicit

Public Sub Example01_BasicRangeQuery()
    On Error GoTo ErrorHandler
    Dim query As SqlQuery
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.SelectColumns("Name", "Salary")
    Set query = query.Where("Department", "=", "IT")
    Set query = query.AndWhere("Salary", ">", 2000)
    Set query = query.OrderBy("Salary", "DESC")
    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "Result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example01 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Name, Salary
FROM Employees
WHERE Department = 'IT' AND Salary > 2000
ORDER BY Salary DESC;
Expected rows: Elena/4200, Ivan/3500, Ivan/3500, Georgi/2500, Emil/2100.

3. Typed conversion and SQL CAST

Run Example03_TypedConversion. WhereAsText and AndWhereAsDouble are VBA equivalents of explicit SQL casts. IGNORE is used because Anna has not-a-number and Viktor has a blank Department.

Option Explicit

Public Sub Example03_TypedConversion()
    On Error GoTo ErrorHandler
    Dim query As SqlQuery
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.SetConversionPolicy("IGNORE")
    Set query = query.SelectColumns("Name", "Salary")
    Set query = query.WhereAsText("Department", "=", "IT")
    Set query = query.AndWhereAsDouble("Salary", ">", 2000)
    Set query = query.OrderByAsDouble("Salary", "DESC")
    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "Typed result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example03 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Name, Salary FROM Employees
WHERE CAST(Department AS TEXT) = 'IT'
  AND CAST(Salary AS DOUBLE) > 2000
ORDER BY CAST(Salary AS DOUBLE) DESC;
Expected rows: Elena/4200, Ivan/3500, Ivan/3500, Georgi/2500, Emil/2100.

4. IN, DISTINCT and paging

Run Example04_InDistinctAndPaging. The query filters departments, converts salary to DOUBLE, removes duplicate projected rows, sorts, and returns five rows.

Option Explicit

Public Sub Example04_InDistinctAndPaging()
    On Error GoTo ErrorHandler
    Dim query As SqlQuery
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.SetConversionPolicy("IGNORE")
    Set query = query.SelectColumns("Name", "Department", "Salary")
    Set query = query.WhereInAsText("Department", "IT", "HR", "Finance")
    Set query = query.AndWhereAsDouble("Salary", ">", 2000)
    Set query = query.Distinct()
    Set query = query.OrderByAsDouble("Salary", "DESC")
    Set query = query.Limit(5)
    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "Five-row result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example04 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT DISTINCT Name, Department, Salary FROM Employees
WHERE Department IN ('IT','HR','Finance')
  AND CAST(Salary AS DOUBLE) > 2000
ORDER BY CAST(Salary AS DOUBLE) DESC LIMIT 5;
Expected rows: Elena, Maria, Ivan, Petar, Georgi.

5. INNER JOIN

Run Example05_InnerJoin. It adds Manager and Office from Departments. Employees without a matching department are excluded.

Option Explicit

Public Sub Example05_InnerJoin()
    On Error GoTo ErrorHandler
    Dim departments As Variant
    Dim query As SqlQuery
    departments = ThisWorkbook.Worksheets("Departments").Range("A1:D5").Value2
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.InnerJoin(departments, "Department", "Department", "TEXT")
    Set query = query.SelectColumns("Name", "Department", "Manager", "Office")
    Set query = query.OrderBy("Name", "ASC")
    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "INNER JOIN result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example05 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT e.Name, e.Department, d.Manager, d.Office
FROM Employees AS e INNER JOIN Departments AS d
ON e.Department = d.Department
ORDER BY e.Name ASC;
Anna and Viktor are excluded because their departments do not match Departments.

Run Example06_LeftJoin. Every employee remains; unmatched right-side columns become Null.

FROM Employees AS e LEFT JOIN Departments AS d
ON e.Department = d.Department

Anna and Viktor remain in the result with Null Manager values.

7. GROUP BY and aggregates

Run Example07_GroupAndAggregate. IGNORE excludes Anna's invalid salary from numeric calculations.

Option Explicit

Public Sub Example07_GroupAndAggregate()
    On Error GoTo ErrorHandler
    Dim query As SqlQuery
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.SetConversionPolicy("IGNORE")
    Set query = query.GroupBy("Department")
    Set query = query.CountRows("EmployeeCount")
    Set query = query.Sum("Salary", "TotalSalary", "DOUBLE")
    Set query = query.Avg("Salary", "AverageSalary", "DOUBLE")
    Set query = query.Min("Salary", "MinSalary", "DOUBLE")
    Set query = query.Max("Salary", "MaxSalary", "DOUBLE")
    query.WriteToRange ThisWorkbook.Worksheets("Results").Range("A1")
    MsgBox "Aggregate result written to Results!A1.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example07 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Department, COUNT(*) AS EmployeeCount,
SUM(CAST(Salary AS DOUBLE)) AS TotalSalary,
AVG(CAST(Salary AS DOUBLE)) AS AverageSalary,
MIN(CAST(Salary AS DOUBLE)) AS MinSalary,
MAX(CAST(Salary AS DOUBLE)) AS MaxSalary
FROM Employees GROUP BY Department;
The blank Department forms a separate group. Anna contributes to COUNT(*) but not numeric salary aggregates.

8. Conversion diagnostics

Run Example08_ConversionDiagnostics. It keeps invalid values in the result and records them in a diagnostic array.

Option Explicit

Public Sub Example08_ConversionDiagnostics()
    On Error GoTo ErrorHandler
    Dim query As SqlQuery
    Dim result As Variant
    Dim errors As Variant
    Set query = New SqlQuery
    Set query = query.LoadRange(ThisWorkbook.Worksheets("Employees").Range("A1:E13"))
    Set query = query.SetConversionPolicy("IGNORE")
    Set query = query.EnableDiagnostics()
    Set query = query.SelectColumns("Name", "Salary")
    Set query = query.OrderByAsDouble("Salary", "ASC")
    result = query.ToArray()
    errors = query.GetConversionErrors()
    ThisWorkbook.Worksheets("Results").Range("A1").Resize(UBound(result, 1), UBound(result, 2)).Value = result
    Debug.Print "Diagnostic records: "; UBound(errors, 1) - 1
    If UBound(errors, 1) > 1 Then Debug.Print "Invalid value: "; errors(2, 5)
    Set query = query.ClearDiagnostics()
    MsgBox "Result written to Results; diagnostics printed in the Immediate window.", vbInformation
    Exit Sub
ErrorHandler:
    MsgBox "Example08 failed: " & Err.Description, vbCritical
End Sub

SQL equivalent

SELECT Name, Salary FROM Employees
ORDER BY CAST(Salary AS DOUBLE) ASC;
-- Diagnostics are application-specific.
The Immediate window reports Anna's not-a-number value.

Execution order

JOIN → WHERE → GROUP BY/aggregates → projection → DISTINCT → ORDER BY → LIMIT/OFFSET.

Current advanced API examples

Logical groups

query.BeginGroup
query.Where "Department", "=", "IT"
query.OrWhere "Department", "=", "HR"
query.EndGroup
query.AndWhere "Active", "=", True

Computed columns

query.SelectColumns "Name"
query.SelectComputed "UPPER(Name)", "NameUpper"
query.AddComputed "LEN(Name)", "NameLength"
query.OrderBy "NameUpper", "ASC"

Composite and chained JOINs

query.InnerJoin lookup, Array("Country", "Code"), Array("Country", "Code"), "TEXT", "Lookup"
query.LeftJoin departments, "Department", "Department", "TEXT", "Dept"
query.LeftJoin offices, "Dept.Code", "Code", "TEXT", "Office"

Aggregate alias ordering

query.GroupBy "Department"
query.CountRows "EmployeeCount"
query.Sum "Salary", "TotalSalary", "DOUBLE"
query.OrderBy "TotalSalary", "DESC"
query.Page 3, 0
